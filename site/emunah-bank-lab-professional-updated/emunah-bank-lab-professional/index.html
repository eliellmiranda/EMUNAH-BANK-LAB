COBOL — EBCLLOAD.cbl
      *===============================================================
      * ARQUIVO      : EBCLLOAD.cbl
      * CAMINHO LOCAL: cobol/batch/EBCLLOAD.cbl
      * HOST / PDS   : Z77948.EMUNAH.DEV.COBOL(EBCLLOAD)
      * FINALIDADE   : Carga inicial dos arquivos seed para os KSDS.
      *
      * FLUXO ESPERADO:
      * 1. Ler CLIENTES.SEQ e CONTAS.SEQ.
      * 2. Montar os registros de negocio.
      * 3. Gravar CLIENTE.KSDS e CONTA.KSDS.
      * 4. Registrar a execucao na auditoria.
      *===============================================================
       IDENTIFICATION DIVISION.
       PROGRAM-ID. EBCLLOAD.

       ENVIRONMENT DIVISION.
       INPUT-OUTPUT SECTION.
       FILE-CONTROL.
      * Entradas seed recebidas do upload via Zowe.
           SELECT CLIENTE-IN ASSIGN TO CLIENTIN.
           SELECT CONTA-IN   ASSIGN TO CONTAIN.

      * Saidas de negocio em VSAM KSDS.
           SELECT CLIENTE-KSDS ASSIGN TO CLIENTE
               ORGANIZATION IS INDEXED
               ACCESS MODE IS DYNAMIC
               RECORD KEY IS CLI-ID.

           SELECT CONTA-KSDS ASSIGN TO CONTA
               ORGANIZATION IS INDEXED
               ACCESS MODE IS DYNAMIC
               RECORD KEY IS CNT-CHAVE.

      * Trilha textual simples da execucao.
           SELECT AUDIT-OUT ASSIGN TO AUDIT.

       DATA DIVISION.
       FILE SECTION.
       FD CLIENTE-IN.
       01 REG-CLIENTE-IN.
          COPY CPCLI001.

       FD CONTA-IN.
       01 REG-CONTA-IN.
          COPY CPCNT001.

       FD CLIENTE-KSDS.
       01 REG-CLIENTE.
          COPY CPCLI001.

       FD CONTA-KSDS.
       01 REG-CONTA.
          COPY CPCNT001.

       FD AUDIT-OUT.
       01 REG-AUDIT PIC X(120).

       WORKING-STORAGE SECTION.
       01 WS-FIM-CLI PIC X VALUE 'N'.
       01 WS-FIM-CNT PIC X VALUE 'N'.

       PROCEDURE DIVISION.
       0000-PRINCIPAL.
      * Para a carga inicial do lab, OPEN OUTPUT recria o conteudo dos
      * KSDS a partir do zero e evita conflitos de reaproveitamento.
           OPEN INPUT CLIENTE-IN CONTA-IN
                OUTPUT CLIENTE-KSDS CONTA-KSDS AUDIT-OUT

      * Carrega o cadastro de clientes.
           PERFORM UNTIL WS-FIM-CLI = 'S'
               READ CLIENTE-IN
                   AT END
                       MOVE 'S' TO WS-FIM-CLI
                   NOT AT END
                       MOVE REG-CLIENTE-IN TO REG-CLIENTE
                       WRITE REG-CLIENTE
               END-READ
           END-PERFORM

      * Carrega o cadastro de contas.
           PERFORM UNTIL WS-FIM-CNT = 'S'
               READ CONTA-IN
                   AT END
                       MOVE 'S' TO WS-FIM-CNT
                   NOT AT END
                       MOVE REG-CONTA-IN TO REG-CONTA
                       WRITE REG-CONTA
               END-READ
           END-PERFORM

      * Auditoria simples de sucesso.
           MOVE 'EBCLLOAD OK - CARGA INICIAL CONCLUIDA' TO REG-AUDIT
           WRITE REG-AUDIT

           CLOSE CLIENTE-IN CONTA-IN
                 CLIENTE-KSDS CONTA-KSDS
                 AUDIT-OUT
           GOBACK.
COBOL — EBVALI01.cbl
      *===============================================================
      * ARQUIVO      : EBVALI01.cbl
      * CAMINHO LOCAL: cobol/batch/EBVALI01.cbl
      * HOST / PDS   : Z77948.EMUNAH.DEV.COBOL(EBVALI01)
      * FINALIDADE   : Validar o lote bruto do dia.
      *
      * ENTRADAS E SAIDAS:
      * ENTRADA  -> DD ENTRADA  -> Z77948.EMUNAH.ARQ.ENTRADA.SEQ
      * VALIDOS  -> DD VALIDOS  -> Z77948.EMUNAH.ARQ.LANCTO.ESDS
      * REJEITOS -> DD REJEITOS -> Z77948.EMUNAH.ARQ.REJEITO.SEQ
      *
      * REGRA DIDATICA:
      * Aceitar somente tipos C e D. Todo o resto vai para rejeito.
      *===============================================================
       IDENTIFICATION DIVISION.
       PROGRAM-ID. EBVALI01.

       ENVIRONMENT DIVISION.
       INPUT-OUTPUT SECTION.
       FILE-CONTROL.
      * Arquivo bruto recebido do dia.
           SELECT MOVTO-IN  ASSIGN TO ENTRADA.
      * Arquivo de saida com movimentos aprovados.
           SELECT MOVTO-VAL ASSIGN TO VALIDOS.
      * Arquivo de saida com movimentos rejeitados.
           SELECT MOVTO-REJ ASSIGN TO REJEITOS.

       DATA DIVISION.
       FILE SECTION.
       FD MOVTO-IN.
       01 REG-MOVTO-IN.
          COPY CPLCT001.

       FD MOVTO-VAL.
       01 REG-MOVTO-VAL.
          COPY CPLCT001.

       FD MOVTO-REJ.
       01 REG-MOVTO-REJ.
          COPY CPLCT001.

       WORKING-STORAGE SECTION.
       01 WS-FIM PIC X VALUE 'N'.

       PROCEDURE DIVISION.
       0000-PRINCIPAL.
      * Le o arquivo bruto e abre os destinos de validos e rejeitos.
           OPEN INPUT MOVTO-IN
                OUTPUT MOVTO-VAL MOVTO-REJ

           PERFORM UNTIL WS-FIM = 'S'
               READ MOVTO-IN
                   AT END
                       MOVE 'S' TO WS-FIM
                   NOT AT END
                       PERFORM 1000-VALIDA
               END-READ
           END-PERFORM

           CLOSE MOVTO-IN MOVTO-VAL MOVTO-REJ
           GOBACK.

       1000-VALIDA.
      * Exemplo didatico: aceita credito e debito.
           IF LCT-TIPO = 'C' OR LCT-TIPO = 'D'
               MOVE REG-MOVTO-IN TO REG-MOVTO-VAL
               WRITE REG-MOVTO-VAL
           ELSE
               MOVE REG-MOVTO-IN TO REG-MOVTO-REJ
               WRITE REG-MOVTO-REJ
           END-IF.
COBOL — EBPOST01.cbl
      *===============================================================
      * ARQUIVO      : EBPOST01.cbl
      * CAMINHO LOCAL: cobol/batch/EBPOST01.cbl
      * HOST / PDS   : Z77948.EMUNAH.DEV.COBOL(EBPOST01)
      * FINALIDADE   : Aplicar os lancamentos validos nas contas.
      *
      * ENTRADAS E SAIDAS:
      * MOVTIN -> DD MOVTIN -> Z77948.EMUNAH.ARQ.LANCTO.ESDS
      * CONTA  -> DD CONTA  -> Z77948.EMUNAH.ARQ.CONTA.KSDS
      * AUDIT  -> DD AUDIT  -> Z77948.EMUNAH.ARQ.AUDIT.SEQ
      *
      * FLUXO ESPERADO:
      * 1. Ler um lancamento aprovado.
      * 2. Montar a chave da conta.
      * 3. Localizar a conta no KSDS.
      * 4. Somar ou subtrair o valor.
      * 5. Regravar a conta e registrar na auditoria.
      *===============================================================
       IDENTIFICATION DIVISION.
       PROGRAM-ID. EBPOST01.

       ENVIRONMENT DIVISION.
       INPUT-OUTPUT SECTION.
       FILE-CONTROL.
           SELECT MOVTO-IN   ASSIGN TO MOVTIN.
           SELECT CONTA-KSDS ASSIGN TO CONTA
               ORGANIZATION IS INDEXED
               ACCESS MODE IS DYNAMIC
               RECORD KEY IS CNT-CHAVE.
           SELECT AUDIT-OUT  ASSIGN TO AUDIT.

       DATA DIVISION.
       FILE SECTION.
       FD MOVTO-IN.
       01 REG-MOVTO-IN.
          COPY CPLCT001.

       FD CONTA-KSDS.
       01 REG-CONTA.
          COPY CPCNT001.

       FD AUDIT-OUT.
       01 REG-AUDIT PIC X(120).

       WORKING-STORAGE SECTION.
       01 WS-FIM PIC X VALUE 'N'.

       PROCEDURE DIVISION.
       0000-PRINCIPAL.
      * MOVTIN contem apenas os movimentos que passaram pela validacao.
           OPEN INPUT MOVTO-IN
                I-O CONTA-KSDS
                EXTEND AUDIT-OUT

           PERFORM UNTIL WS-FIM = 'S'
               READ MOVTO-IN
                   AT END
                       MOVE 'S' TO WS-FIM
                   NOT AT END
                       PERFORM 1000-POSTA
               END-READ
           END-PERFORM

           CLOSE MOVTO-IN CONTA-KSDS AUDIT-OUT
           GOBACK.

       1000-POSTA.
      * Monta a chave da conta a partir do movimento.
           MOVE LCT-AGENCIA TO CNT-AGENCIA
           MOVE LCT-CONTA   TO CNT-NUM-CONTA

           READ CONTA-KSDS
               INVALID KEY
                   MOVE 'CONTA NAO ENCONTRADA NO POST' TO REG-AUDIT
                   WRITE REG-AUDIT
               NOT INVALID KEY
                   IF LCT-TIPO = 'C'
      * Credito soma ao saldo atual.
                       ADD LCT-VALOR TO CNT-SALDO-ATUAL
                   ELSE
      * Debito subtrai do saldo atual.
                       SUBTRACT LCT-VALOR FROM CNT-SALDO-ATUAL
                   END-IF

      * Persiste a nova imagem da conta.
                   REWRITE REG-CONTA

      * Auditoria resumida da aplicacao.
                   MOVE 'POST OK - MOVIMENTO APLICADO' TO REG-AUDIT
                   WRITE REG-AUDIT
           END-READ.

O que é o Zowe CLI?

O Zowe CLI é uma ferramenta open-source que permite interagir com o mainframe z/OS a partir do terminal do seu computador local, sem precisar de emulador 3270.

Instalação

npm install -g @zowe/cli
zowe --version

Configuração de Perfil

# Criar perfil z/OSMF
zowe profiles create zosmf meu-perfil \
  --host meu-mainframe.empresa.com \
  --port 443 \
  --user MEUSUSR \
  --password MINHASENHA \
  --reject-unauthorized false

# Definir como padrão
zowe profiles set-default zosmf meu-perfil

7.1 Job com ABEND — Como Investigar

Passo a Passo

1. Verificar o ABEND CODE no output do job
   → Exemplo: "ABEND S0C7 IN STEP STEP010"

2. Localizar o step que abendou
   → Ver JESMSGLG ou JESJCL para identificar o step

3. Verificar SYSOUT do step
   → Mensagens do programa, display, WTO

4. Verificar CEEDUMP ou SYSUDUMP (se disponível)
   → Registros de memória e trace do programa

5. Consultar tabela de ABENDs (ver Seção 8)
   → Entender o significado do código

6. Analisar dados de entrada
   → Verificar se o arquivo de entrada está correto

7. Corrigir e re-submeter

Checklist de investigação

  • [ ] Qual é o código ABEND? (Sxxx ou Uxxx)
  • [ ] Em qual step ocorreu?
  • [ ] Qual programa estava executando?
  • [ ] Há mensagem de erro no SYSOUT?
  • [ ] O arquivo de entrada existe e está correto?
  • [ ] Há CEEDUMP ou SYSUDUMP?
  • [ ] O programa foi compilado recentemente? Houve mudança?

//STEP010  EXEC PGM=SORT,...
//STEP020  EXEC PGM=MEUPROG,COND=(8,LT,STEP010)
*          ↑ Só executa se RC do STEP010 for MENOR que 8

Lógica do parâmetro COND

COND=(valor,operador,stepname)

Operadores: LT, LE, EQ, NE, GT, GE
Leitura: "PULAR este step SE condição for verdadeira"

COND=(4,LT,STEP010) = Pular se 4 < RC(STEP010) = Pular se RC > 4

Sintomas

  • Job fica em estado WAIT ou HELD
  • Mensagem: IEF861I JOBNAME WAITING FOR DATA SET

Investigação via SDSF

=SD → RES (Resource)
Verificar quem está usando o dataset

Via Zowe CLI

zowe zos-console issue cmd "D GRS,RES=(DATASET,'MY.LOCKED.DS')"

Soluções

  1. Aguardar o job que tem o dataset terminar
  2. Cancelar o job bloqueante (se autorizado)
  3. Alterar DISP de OLD para SHR (se possível)
  4. Verificar se há jobs em abend segurando o dataset

7.4 Erro de SQL (SQLCODE) — Interpretação

Fluxo de investigação de erro SQL

1. Verificar SQLCODE retornado
   → Positivo = Warning, Negativo = Erro

2. Verificar SQLERRM
   → Mensagem descritiva do erro

3. Consultar tabela de SQLCODEs (Seção 3.1)

4. Verificar dados de entrada
   → Chave duplicada? Dado nulo? Tipo errado?

5. Verificar se objeto existe
   → Tabela, View, Index — usar LISTCAT ou SYSIBM.SYSTABLES

6. Verificar autorização
   → GRANT SELECT/INSERT/UPDATE/DELETE na tabela?

No programa COBOL — verificar SQLCODE

EXEC SQL
  UPDATE DB2ADM.CLIENTES
     SET SALDO = :WS-SALDO
   WHERE NR_CONTA = :WS-CONTA
END-EXEC.

EVALUATE TRUE
  WHEN SQLCODE = 0
    CONTINUE
  WHEN SQLCODE = 100
    PERFORM REGISTRO-NAO-ENCONTRADO
  WHEN SQLCODE < 0
    MOVE SQLCODE TO WS-SQLCODE-DISP
    PERFORM ERRO-SQL
END-EVALUATE.

8.5 Como Ler um DUMP

Estrutura do SYSUDUMP / CEEDUMP

1. HEADER — Informações do job (jobname, stepname, data/hora)
2. ABEND CODE — Código do abend (S0C7, U0000, etc.)
3. PSW — Program Status Word (endereço da instrução com problema)
4. GPR — General Purpose Registers (R0 a R15)
5. STORAGE DUMP — Conteúdo da memória (hexadecimal + EBCDIC)
6. TRACEBACK — Pilha de chamadas (quais programas/parágrafos)

Dica prática para S0C7

1. Localizar PSW no dump → tem o endereço que causou o abend
2. Correlacionar com o mapa de linkagem do compilador (SYSMAP)
3. Identificar o parágrafo/instrução COBOL
4. Verificar qual campo estava com dado inválido
5. Inspecionar o STORAGE DUMP naquela posição
   → Dado deve ser numérico (0-9 em EBCDIC = X'F0'-X'F9')
   → Se contiver X'40' (espaço) em campo PIC 9 = causa do S0C7

11.1 Movimentação de Dados

MOVE — Atribuir valores

O verbo mais usado em qualquer programa COBOL. Copia o conteúdo de um campo para outro.

MOVE WS-SALDO-ATUAL    TO WS-SALDO-ANTERIOR
MOVE ZEROS             TO WS-CONTADOR
MOVE SPACES            TO WS-NOME-CLIENTE
MOVE 'S'               TO WS-FLAG-PROCESSADO
MOVE WS-DATA-SISTEMA   TO WS-DATA-MOVIMENTO
MOVE FUNCTION CURRENT-DATE TO WS-DATA-HORA

Em bancos: usado intensamente para montar registros de saída, preencher campos de layout de remessa/retorno (CNAB 240, CNAB 400), e preparar variáveis para SQL.

MOVE CORRESPONDING — Mover campos com mesmo nome

Move automaticamente campos que têm o mesmo nome entre dois grupos.

01 WS-CLIENTE-ENTRADA.
   05 NR-CONTA  PIC 9(10).
   05 NOME      PIC X(30).
   05 SALDO     PIC 9(13)V99.

01 WS-CLIENTE-SAIDA.
   05 NR-CONTA  PIC 9(10).
   05 NOME      PIC X(30).
   05 SALDO     PIC 9(13)V99.

* Move todos os campos com mesmo nome de uma vez
MOVE CORRESPONDING WS-CLIENTE-ENTRADA TO WS-CLIENTE-SAIDA

Em bancos: útil para copiar registros entre áreas de trabalho com layouts similares.

STRING — Concatenar campos

Junta vários campos em um único campo destino.

STRING WS-DDD     DELIMITED BY SPACE
       '-'        DELIMITED BY SIZE
       WS-FONE    DELIMITED BY SPACE
       INTO WS-FONE-FORMATADO

* Resultado: "11-999887766"

Em bancos: formatação de mensagens de log, montagem de chave composta, geração de descrição de movimento.

UNSTRING — Separar campos

Divide um campo em partes baseado em delimitador.

UNSTRING WS-LINHA-RETORNO
    DELIMITED BY ';'
    INTO WS-CAMPO1
         WS-CAMPO2
         WS-CAMPO3
    TALLYING IN WS-QTD-CAMPOS

Em bancos: leitura de arquivos de retorno com campos delimitados.

INSPECT — Varrer e substituir caracteres

Conta ou substitui ocorrências de caracteres em um campo.

* Contar espaços
INSPECT WS-CAMPO
    TALLYING WS-CONTADOR FOR ALL SPACES

* Substituir vírgula por ponto (para valor monetário)
INSPECT WS-VALOR-STRING
    CONVERTING ',' TO '.'

* Substituir espaços por zeros
INSPECT WS-CAMPO-NUMERICO
    CONVERTING SPACES TO ZEROS

Em bancos: limpeza de dados de entrada, normalização de campos numéricos recebidos de arquivos externos.

11.2 Aritmética e Cálculos Financeiros

ADD — Somar

ADD WS-VALOR-CREDITO  TO WS-SALDO-CONTA
ADD WS-JUROS          TO WS-VALOR-PARCELA
ADD 1                 TO WS-CONTADOR
ADD WS-VL1 WS-VL2    TO WS-TOTAL

* Com GIVING (resultado em outro campo — não altera os originais)
ADD WS-PRINCIPAL WS-JUROS GIVING WS-VALOR-TOTAL

SUBTRACT — Subtrair

SUBTRACT WS-VALOR-DEBITO   FROM WS-SALDO-CONTA
SUBTRACT WS-DESCONTO       FROM WS-VALOR-BRUTO GIVING WS-VALOR-LIQUIDO
SUBTRACT 1                 FROM WS-CONTADOR

MULTIPLY — Multiplicar

MULTIPLY WS-TAXA-JUROS BY WS-SALDO GIVING WS-VALOR-JUROS
MULTIPLY 12            BY WS-PRESTACAO GIVING WS-TOTAL-ANUAL

DIVIDE — Dividir

DIVIDE WS-TOTAL-DIVIDA BY WS-QTD-PARCELAS
    GIVING WS-VALOR-PARCELA
    REMAINDER WS-RESTO

* Sempre verificar divisão por zero antes!
IF WS-QTD-PARCELAS > ZEROS
    DIVIDE WS-TOTAL BY WS-QTD-PARCELAS GIVING WS-PARCELA
END-IF

COMPUTE — Expressões matemáticas complexas

O mais poderoso para cálculos financeiros. Suporta expressões completas.

* Juros compostos
COMPUTE WS-MONTANTE =
    WS-PRINCIPAL * ((1 + WS-TAXA) ** WS-PERIODO)

* Valor presente
COMPUTE WS-VALOR-PRESENTE =
    WS-FLUXO-FUTURO / ((1 + WS-TAXA-DESCONTO) ** WS-PERIODO)

* Parcela de financiamento (Price)
COMPUTE WS-PARCELA ROUNDED =
    (WS-PV * WS-TAXA) /
    (1 - (1 + WS-TAXA) ** (-WS-N))

* IOF diário
COMPUTE WS-IOF =
    WS-VALOR-OPERACAO * WS-ALIQ-IOF * WS-DIAS-CORRIDOS

Em bancos: cálculo de juros, IOF, CET, amortização, rentabilidade de investimentos — tudo passa pelo COMPUTE.

ON SIZE ERROR — Proteger contra overflow numérico

COMPUTE WS-RESULTADO = WS-VALOR1 * WS-VALOR2
    ON SIZE ERROR
        MOVE 'OVERFLOW' TO WS-STATUS
        PERFORM ROT-ERRO-CALCULO
    NOT ON SIZE ERROR
        PERFORM ROT-GRAVA-RESULTADO
END-COMPUTE

11.3 Controle de Fluxo

IF / ELSE / END-IF — Desvio condicional

IF WS-SALDO >= WS-VALOR-SAQUE
    PERFORM ROT-APROVAR-SAQUE
    ADD WS-VALOR-SAQUE TO WS-TOTAL-SAQUES
ELSE
    MOVE 'SALDO INSUFICIENTE' TO WS-MSG-RETORNO
    PERFORM ROT-RECUSAR-SAQUE
END-IF

* Condição composta
IF WS-TIPO-CONTA = 'CC'
   AND WS-STATUS-CONTA = 'A'
   AND WS-SALDO > ZEROS
    PERFORM ROT-PROCESSA
END-IF

* Verificação de nulo/espaço
IF WS-NOME = SPACES OR WS-NOME = LOW-VALUES
    PERFORM ROT-DADO-INVALIDO
END-IF

EVALUATE — Equivalente ao SWITCH/CASE

O mais elegante para múltiplas condições — muito usado em bancos para tipos de movimento.

EVALUATE WS-TIPO-MOVIMENTO
    WHEN 'PIX'    PERFORM ROT-PROCESSA-PIX
    WHEN 'TED'    PERFORM ROT-PROCESSA-TED
    WHEN 'DOC'    PERFORM ROT-PROCESSA-DOC
    WHEN 'BOL'    PERFORM ROT-PROCESSA-BOLETO
    WHEN 'DEB'    PERFORM ROT-PROCESSA-DEBITO
    WHEN OTHER    PERFORM ROT-TIPO-INVALIDO
END-EVALUATE

* EVALUATE com múltiplos valores
EVALUATE WS-CODIGO-RETORNO
    WHEN 0        PERFORM ROT-SUCESSO
    WHEN 1 THRU 9 PERFORM ROT-WARNING
    WHEN NEGATIVE PERFORM ROT-ERRO
    WHEN OTHER    PERFORM ROT-DESCONHECIDO
END-EVALUATE

* EVALUATE TRUE — para condições complexas
EVALUATE TRUE
    WHEN WS-SALDO > 100000
        PERFORM ROT-CLIENTE-PREMIUM
    WHEN WS-SALDO > 10000
        PERFORM ROT-CLIENTE-PLUS
    WHEN WS-SALDO > 0
        PERFORM ROT-CLIENTE-BASICO
    WHEN OTHER
        PERFORM ROT-CONTA-NEGATIVA
END-EVALUATE

PERFORM — Executar parágrafo / laço

* Chamar parágrafo uma vez
PERFORM ROT-VALIDA-DADOS
PERFORM ROT-GRAVA-MOVIMENTO

* Laço com condição (UNTIL)
PERFORM ROT-LER-ARQUIVO
    UNTIL WS-FIM-ARQUIVO = 'S'

* Laço com contador (VARYING)
PERFORM VARYING WS-IDX FROM 1 BY 1
    UNTIL WS-IDX > WS-QTD-PARCELAS
        PERFORM ROT-CALCULAR-PARCELA
END-PERFORM

* PERFORM THRU — executa sequência de parágrafos
PERFORM ROT-INICIO THRU ROT-INICIO-FIM

* PERFORM com TIMES
PERFORM ROT-TENTATIVA 3 TIMES

Em bancos: estrutura de todos os programas batch — laço principal de leitura, processamento e escrita de movimentos.

11.4 Operações de Arquivo

OPEN — Abrir arquivo

OPEN INPUT  ARQ-MOVIMENTOS        *> Somente leitura
OPEN OUTPUT ARQ-REMESSA           *> Somente escrita (cria/substitui)
OPEN I-O    ARQ-CLIENTES-VSAM     *> Leitura e atualização (VSAM)
OPEN EXTEND ARQ-LOG               *> Acrescenta ao final

READ — Ler registro

* Leitura sequencial
READ ARQ-MOVIMENTOS
    AT END MOVE 'S' TO WS-FIM-ARQUIVO
    NOT AT END PERFORM ROT-PROCESSA-REGISTRO
END-READ

* Leitura VSAM por chave (RANDOM)
MOVE WS-NR-CONTA TO NR-CONTA-VSAM
READ ARQ-CLIENTES-VSAM
    INVALID KEY PERFORM ROT-CLIENTE-NAO-ENCONTRADO
    NOT INVALID KEY PERFORM ROT-PROCESSA-CLIENTE
END-READ

* Verificar status após leitura
READ ARQ-ENTRADA INTO WS-REGISTRO
EVALUATE WS-FILE-STATUS
    WHEN '00' CONTINUE
    WHEN '10' MOVE 'S' TO WS-FIM-ARQUIVO
    WHEN OTHER PERFORM ROT-ERRO-LEITURA
END-EVALUATE

WRITE — Gravar registro

* Escrita sequencial
WRITE REG-SAIDA FROM WS-REGISTRO-SAIDA

* Com controle de formulário (relatórios)
WRITE REG-RELATORIO FROM WS-LINHA
    AFTER ADVANCING 1 LINE
WRITE REG-RELATORIO FROM WS-CABECALHO
    AFTER ADVANCING PAGE

* VSAM — novo registro
WRITE REG-CLIENTES FROM WS-NOVO-CLIENTE
    INVALID KEY PERFORM ROT-CHAVE-DUPLICADA
END-WRITE

REWRITE — Atualizar registro (VSAM)

* Atualizar registro lido anteriormente
MOVE WS-NOVO-SALDO TO SALDO-VSAM
REWRITE REG-CLIENTES FROM WS-CLIENTE-ATUALIZADO
    INVALID KEY PERFORM ROT-ERRO-ATUALIZACAO
END-REWRITE

DELETE — Deletar registro VSAM

DELETE ARQ-CLIENTES-VSAM
    INVALID KEY PERFORM ROT-REGISTRO-NAO-EXISTE
END-DELETE

CLOSE — Fechar arquivo

CLOSE ARQ-MOVIMENTOS

      ARQ-CLIENTES-VSAM

11.5 Acesso a Banco de Dados DB2

EXEC SQL SELECT — Consulta única

EXEC SQL
    SELECT NR_CONTA,
           NOME_CLIENTE,
           VL_SALDO,
           DT_ABERTURA
      INTO :WS-NR-CONTA,
           :WS-NOME-CLIENTE,
           :WS-VL-SALDO,
           :WS-DT-ABERTURA
      FROM DB2ADM.TB_CLIENTES
     WHERE NR_CONTA = :WS-CONTA-PESQ
       AND CD_STATUS = 'A'
END-EXEC

EVALUATE SQLCODE
    WHEN 0   CONTINUE
    WHEN 100 PERFORM ROT-CONTA-NAO-ENCONTRADA
    WHEN OTHER PERFORM ROT-ERRO-SQL
END-EVALUATE

EXEC SQL CURSOR — Consulta múltiplos registros

* 1. Declarar cursor (em WORKING-STORAGE ou início do programa)
EXEC SQL
    DECLARE CUR-MOVIMENTOS CURSOR FOR
    SELECT NR_MOVIMENTO,
           VL_MOVIMENTO,
           DT_MOVIMENTO,
           TP_MOVIMENTO
      FROM DB2ADM.TB_MOVIMENTOS
     WHERE NR_CONTA  = :WS-NR-CONTA
       AND DT_MOVIMENTO BETWEEN :WS-DT-INI AND :WS-DT-FIM
     ORDER BY DT_MOVIMENTO ASC
END-EXEC.

* 2. Abrir cursor
EXEC SQL OPEN CUR-MOVIMENTOS END-EXEC.

* 3. Ler registros em laço
PERFORM UNTIL SQLCODE = 100
    EXEC SQL
        FETCH CUR-MOVIMENTOS
        INTO :WS-NR-MOVIMENTO,
             :WS-VL-MOVIMENTO,
             :WS-DT-MOVIMENTO,
             :WS-TP-MOVIMENTO
    END-EXEC

    IF SQLCODE = 0
        PERFORM ROT-PROCESSA-MOVIMENTO
    END-IF
END-PERFORM.

* 4. Fechar cursor
EXEC SQL CLOSE CUR-MOVIMENTOS END-EXEC.

EXEC SQL INSERT — Incluir registro

EXEC SQL
    INSERT INTO DB2ADM.TB_MOVIMENTOS
           (NR_CONTA,
            VL_MOVIMENTO,
            TP_MOVIMENTO,
            DT_MOVIMENTO,
            NR_SEQUENCIA)
    VALUES (:WS-NR-CONTA,
            :WS-VL-MOVIMENTO,
            :WS-TP-MOVIMENTO,
            CURRENT DATE,
            NEXT VALUE FOR DB2ADM.SEQ_MOVIMENTO)
END-EXEC

IF SQLCODE NOT = 0
    PERFORM ROT-ERRO-INSERT
END-IF

EXEC SQL UPDATE — Atualizar registro

EXEC SQL
    UPDATE DB2ADM.TB_CLIENTES
       SET VL_SALDO      = VL_SALDO + :WS-VL-CREDITO,
           DT_ULT_MOVTO  = CURRENT DATE,
           TS_ATUALIZACAO = CURRENT TIMESTAMP
     WHERE NR_CONTA = :WS-NR-CONTA
       AND CD_STATUS = 'A'
END-EXEC

EVALUATE SQLCODE
    WHEN 0
        MOVE SQLCODE TO WS-OK
    WHEN 100
        PERFORM ROT-CONTA-NAO-ENCONTRADA
    WHEN OTHER
        PERFORM ROT-ERRO-UPDATE
END-EVALUATE

EXEC SQL DELETE — Deletar registro

EXEC SQL
    DELETE FROM DB2ADM.TB_MOVIMENTOS_TEMP
     WHERE NR_CONTA      = :WS-NR-CONTA
       AND DT_MOVIMENTO   < (CURRENT DATE - 30 DAYS)
       AND CD_STATUS      = 'P'
END-EXEC

EXEC SQL COMMIT / ROLLBACK — Controle de transação

* Confirmar todas as alterações
EXEC SQL COMMIT END-EXEC

* Desfazer todas as alterações desde o último COMMIT
EXEC SQL ROLLBACK END-EXEC

* Padrão em programas bancários batch:
PERFORM ROT-PROCESSA-LOTE
IF WS-ERRO-LOTE = 'N'
    EXEC SQL COMMIT END-EXEC
ELSE
    EXEC SQL ROLLBACK END-EXEC
    PERFORM ROT-GRAVA-LOG-ERRO
END-IF

Em bancos: COMMIT e ROLLBACK são críticos. Um programa de débito/crédito sempre precisa garantir que ambas as operações sejam feitas juntas — ou as duas, ou nenhuma.

STOP RUN / GOBACK — Encerrar programa

* Encerra o programa e retorna ao sistema operacional (JCL)
STOP RUN

* Encerra subprograma e retorna ao chamador (mais correto para subprogramas)
GOBACK

* Em bancos: sempre fazer COMMIT ou ROLLBACK antes de encerrar
PERFORM ROT-FINALIZAR
EXEC SQL COMMIT END-EXEC
STOP RUN

SET — Trabalhar com índices e switches

* Definir índice de tabela
SET WS-IDX TO 1
SET WS-IDX UP BY 1

* Condição 88 (flag booleano)
01 WS-STATUS-CONTA.

      88 CONTA-ATIVA   VALUE 'A'.
      88 CONTA-INATIVA VALUE 'I'.

PROCEDURE DIVISION.

 0000-PRINCIPAL.
     PERFORM 1000-INICIAR
     PERFORM 2000-PROCESSAR UNTIL FIM-ARQUIVO
     PERFORM 3000-FINALIZAR
     STOP RUN.

     OPEN INPUT ARQ-MOVIMENTOS
     IF WS-STATUS-MOVTOS NOT = '00'
         DISPLAY 'ERRO AO ABRIR ARQUIVO: ' WS-STATUS-MOVTOS
         MOVE 8 TO RETURN-CODE
         STOP RUN
     END-IF
     PERFORM 1100-LER-PROXIMO.

 1100-LER-PROXIMO.
     READ ARQ-MOVIMENTOS
         AT END SET FIM-ARQUIVO TO TRUE
     END-READ.

 2000-PROCESSAR.
     ADD 1 TO WS-QTD-LIDOS
     EVALUATE MV-TP-MOVIMENTO
         WHEN 'CR'
             PERFORM 2100-PROCESSAR-CREDITO
         WHEN 'DB'
             PERFORM 2200-PROCESSAR-DEBITO
         WHEN OTHER
             ADD 1 TO WS-QTD-ERROS
     END-EVALUATE
     PERFORM 1100-LER-PROXIMO.

 2100-PROCESSAR-CREDITO.

         UPDATE DB2ADM.TB_CLIENTES

                DT_ULT_MOVTO = CURRENT DATE

         ADD MV-VL-MOVIMENTO TO WS-VL-TOTAL-CR
         EXEC SQL COMMIT END-EXEC
     ELSE
         ADD 1 TO WS-QTD-ERROS
         EXEC SQL ROLLBACK END-EXEC
     END-IF.

 2200-PROCESSAR-DEBITO.
     EXEC SQL
         UPDATE DB2ADM.TB_CLIENTES
            SET VL_SALDO = VL_SALDO - :MV-VL-MOVIMENTO,
                DT_ULT_MOVTO = CURRENT DATE
          WHERE NR_CONTA  = :MV-NR-CONTA
            AND VL_SALDO >= :MV-VL-MOVIMENTO
     END-EXEC
     EVALUATE SQLCODE
         WHEN 0
             ADD 1              TO WS-QTD-DEBITOS
             ADD MV-VL-MOVIMENTO TO WS-VL-TOTAL-DB
             EXEC SQL COMMIT END-EXEC
         WHEN 100
             DISPLAY 'SALDO INSUF CONTA: ' MV-NR-CONTA
             ADD 1 TO WS-QTD-ERROS
             EXEC SQL ROLLBACK END-EXEC
         WHEN OTHER
             ADD 1 TO WS-QTD-ERROS
             EXEC SQL ROLLBACK END-EXEC
     END-EVALUATE.

 3000-FINALIZAR.
     CLOSE ARQ-MOVIMENTOS
     DISPLAY '*** RELATORIO DE PROCESSAMENTO ***'
     DISPLAY 'REGISTROS LIDOS   : ' WS-QTD-LIDOS
     DISPLAY 'CREDITOS OK       : ' WS-QTD-CREDITOS
     DISPLAY 'DEBITOS OK        : ' WS-QTD-DEBITOS
     DISPLAY 'ERROS             : ' WS-QTD-ERROS

     DISPLAY 'TOTAL DEBITOS     : ' WS-VL-TOTAL-DB
     IF WS-QTD-ERROS > ZEROS
         MOVE 4 TO RETURN-CODE
     END-IF.
