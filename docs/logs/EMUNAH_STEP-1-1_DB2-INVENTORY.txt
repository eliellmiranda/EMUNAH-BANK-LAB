 COMMAND

 192.168.100.150:8102) or via TSO SPUFI / DSNTEP2 on z/OS.

 -- -------------------------------------------------------
 -- STEP 1.1A — List all tables in schema EMUNAH
 -- -------------------------------------------------------
 SELECT TBNAME       AS TABLE_NAME,
        CREATOR      AS SCHEMA,
        CARD         AS ROW_COUNT_ESTIMATE,
        TSNAME       AS TABLESPACE
 FROM   SYSIBM.SYSTABLES
 WHERE  CREATOR = 'EMUNAH'
 ORDER  BY TBNAME;

 -- -------------------------------------------------------
 -- STEP 1.1B — List all columns for every EMUNAH table
 -- -------------------------------------------------------
 SELECT TBNAME        AS TABLE_NAME,
        COLNO         AS COL_NUM,
        NAME          AS COLUMN_NAME,
        COLTYPE       AS DATA_TYPE,
        LENGTH,
        SCALE,
        NULLS         AS NULLABLE,
        DEFAULT       AS DEFAULT_VALUE
 FROM   SYSIBM.SYSCOLUMNS
 WHERE  TBCREATOR = 'EMUNAH'
 ORDER  BY TBNAME, COLNO;

 -- -------------------------------------------------------
 -- STEP 1.1C — List all constraints (PK, FK, CHECK)
 -- -------------------------------------------------------
 SELECT TBNAME     AS TABLE_NAME,
        CONSTNAME  AS CONSTRAINT_NAME,
        TYPE       AS CONSTRAINT_TYPE
 FROM   SYSIBM.SYSTABCONST
 WHERE  TBCREATOR = 'EMUNAH'
 ORDER  BY TBNAME, TYPE;
